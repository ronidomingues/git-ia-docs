    \foreach \l/\x in {X/0,Y/1.9}{\node[commit=white, fill=azulMB, font=\small\bfseries\color{white}] (\l) at (\x,-0.9) {\l};}
    \draw[hist=white] (X)--(Y);
    \node[commitm] (M) at (5.6,0) {M};
    \draw[hist=verdeMB] (D) to[out=0,in=135] (M);
    \draw[hist=white] (Y) to[out=0,in=-135] (M);
  \end{scope}}
